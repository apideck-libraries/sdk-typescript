# FiscalPeriod

## Example Usage

```typescript
import { FiscalPeriod } from "@apideck/unify/models/components";

let value: FiscalPeriod = {
  id: "12345",
  name: "Jan 2026",
  fiscalYear: "2026",
  periodNumber: 1,
  startDate: new Date("2026-01-01"),
  endDate: new Date("2026-01-31"),
  status: "open",
  lockedThrough: new Date("2026-09-30"),
  type: "period",
  isAdjustment: false,
  createdAt: new Date("2020-09-30T07:43:32.000Z"),
  updatedAt: new Date("2020-09-30T07:43:32.000Z"),
};
```
