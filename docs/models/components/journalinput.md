# JournalInput

## Example Usage

```typescript
import { JournalInput } from "@apideck/unify/models/components";

let value: JournalInput = {
  code: "90",
  name: "General journal",
  description: "General journal",
  type: "general",
  allowVat: true,
  currency: "USD",
  iban: "GB33BUKB20201555555555",
  defaultAccount: {
    id: "123456",
    type: "ledger_account",
    code: "1100",
    displayId: "123456",
    accountNumber: "123465",
  },
  blocked: false,
};
```
